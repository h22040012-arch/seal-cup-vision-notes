cmake_minimum_required(VERSION 3.20)
project(SealCupInference LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Native integration requires Windows x64")
endif()
set(ONNXRUNTIME_INCLUDE_DIR "" CACHE PATH "ONNX Runtime 1.20.1 include directory")
if(NOT EXISTS "${ONNXRUNTIME_INCLUDE_DIR}/onnxruntime_c_api.h")
  message(FATAL_ERROR "Set ONNXRUNTIME_INCLUDE_DIR to ONNX Runtime 1.20.1 headers")
endif()
add_library(CupVmInference20261030 SHARED cup_native_top1.cpp)
target_compile_features(CupVmInference20261030 PRIVATE cxx_std_17)
target_include_directories(CupVmInference20261030 PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(CupVmInference20261030 PRIVATE advapi32)
